document.addEventListener('DOMContentLoaded', function() {
    // Logout functionality
    const statusBadge = document.querySelector('.status-badge');
    if (statusBadge) {
        statusBadge.addEventListener('click', function() {
            fetch('/admin/logout', { method: 'POST' }).then(function() { location.href = '/admin/login'; });
        });
    }

    // Refresh functionality
    const refreshBtn = document.getElementById('refresh-btn');
    if (refreshBtn) {
        refreshBtn.addEventListener('click', function() {
            location.reload();
        });
    }

    // Setup Telegram Webhook
    const tgBtn = document.getElementById('setup-telegram-btn');
    if (tgBtn) {
        tgBtn.addEventListener('click', function() {
            tgBtn.innerText = 'Setting...';
            fetch('/admin/setup-telegram', { method: 'POST' })
                .then(function(res) { return res.json(); })
                .then(function(data) { 
                    alert('Telegram Webhook: ' + (data.ok ? 'Success' : 'Failed - ' + data.description)); 
                    tgBtn.innerText = 'Setup Telegram Webhook'; 
                })
                .catch(function(err) { 
                    alert('Error: ' + err); 
                    tgBtn.innerText = 'Setup Telegram Webhook'; 
                });
        });
    }

    // Setup Meta Subscriptions
    const metaBtn = document.getElementById('setup-meta-btn');
    if (metaBtn) {
        metaBtn.addEventListener('click', function() {
            metaBtn.innerText = 'Subscribing...';
            fetch('/admin/setup-meta', { method: 'POST' })
                .then(function(res) { return res.json(); })
                .then(function(data) { 
                    alert('Meta Subscribed: ' + (data.success ? 'Success' : JSON.stringify(data))); 
                    metaBtn.innerText = 'Subscribe Page to App'; 
                })
                .catch(function(err) { 
                    alert('Error: ' + err); 
                    metaBtn.innerText = 'Subscribe Page'; 
                });
        });
    }

    // General test handler creator
    function createTestHandler(btnId, inputId, endpoint) {
        const btn = document.getElementById(btnId);
        if (btn) {
            btn.addEventListener('click', function() {
                const input = document.getElementById(inputId);
                const id = input ? input.value : null;
                if (!id) return alert('Please enter an ID first.');
                
                const originalText = btn.innerText;
                btn.innerText = 'Sending...';
                btn.disabled = true;

                fetch(endpoint, { 
                    method: 'POST', 
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ recipientId: id })
                })
                .then(function(res) { return res.json(); })
                .then(function(data) {
                    alert('Status: ' + (data.success ? 'Success' : 'Error: ' + JSON.stringify(data)));
                })
                .catch(function(err) {
                    alert('Error: ' + err);
                })
                .finally(function() {
                    btn.innerText = originalText;
                    btn.disabled = false;
                });
            });
        }
    }

    // Test Telegram bot
    const tgTestBtn = document.getElementById('test-telegram-btn');
    if (tgTestBtn) {
        tgTestBtn.addEventListener('click', function() {
            const originalText = tgTestBtn.innerText;
            tgTestBtn.innerText = 'Sending...';
            tgTestBtn.disabled = true;

            fetch('/admin/test-telegram', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({})
            })
            .then(function(res) { return res.json(); })
            .then(function(data) {
                alert('Status: ' + (data.success ? 'Success' : 'Error: ' + JSON.stringify(data)));
            })
            .catch(function(err) {
                alert('Error: ' + err);
            })
            .finally(function() {
                tgTestBtn.innerText = originalText;
                tgTestBtn.disabled = false;
            });
        });
    }

    createTestHandler('test-meta-btn', 'test-meta-id', '/admin/test-meta');
    createTestHandler('test-insta-btn', 'test-insta-id', '/admin/test-insta');
    createTestHandler('test-whatsapp-btn', 'test-whatsapp-id', '/admin/test-whatsapp');

    // Telegram Account Login logic
    const tgSendBtn = document.getElementById('tg-send-code');
    const tgVerifyBtn = document.getElementById('tg-verify-code');
    const tgQrBtn = document.getElementById('tg-qr-login');
    const tgPhoneInput = document.getElementById('tg-phone');
    const tgCodeInput = document.getElementById('tg-code');
    const tgStep1 = document.getElementById('tg-auth-step-1');
    const tgStep2 = document.getElementById('tg-auth-step-2');
    const tgQrContainer = document.getElementById('tg-qr-container');
    const tgQrBack = document.getElementById('tg-qr-back');
    const tgQrPlaceholder = document.getElementById('tg-qr-placeholder');

    var qrPollInterval = null;
    var qrExpiryTimeout = null;

    function stopQrPolling() {
        if (qrPollInterval) { clearInterval(qrPollInterval); qrPollInterval = null; }
        if (qrExpiryTimeout) { clearTimeout(qrExpiryTimeout); qrExpiryTimeout = null; }
    }

    function showPhoneForm() {
        stopQrPolling();
        tgQrContainer.style.display = 'none';
        tgStep1.style.display = 'block';
        tgStep2.style.display = 'none';
        if (tgQrPlaceholder) tgQrPlaceholder.innerHTML = 'QR LOADING...';
    }

    function renderQrCode(url) {
        if (!tgQrPlaceholder) return;
        tgQrPlaceholder.innerHTML = '';
        // Load QRCode.js from CDN dynamically
        if (typeof QRCode !== 'undefined') {
            new QRCode(tgQrPlaceholder, {
                text: url,
                width: 128,
                height: 128,
                colorDark: '#000000',
                colorLight: '#ffffff',
                correctLevel: QRCode.CorrectLevel.M
            });
        } else {
            // Fallback: load script then render
            var script = document.createElement('script');
            script.src = 'https://cdnjs.cloudflare.com/ajax/libs/qrcodejs/1.0.0/qrcode.min.js';
            script.onload = function() {
                new QRCode(tgQrPlaceholder, {
                    text: url,
                    width: 128,
                    height: 128,
                    colorDark: '#000000',
                    colorLight: '#ffffff',
                    correctLevel: QRCode.CorrectLevel.M
                });
            };
            document.head.appendChild(script);
        }
    }

    function startQrPolling(token) {
        stopQrPolling();

        // Poll every 2 seconds
        qrPollInterval = setInterval(function() {
            fetch('/admin/tg-qr-status?token=' + token)
                .then(function(r) { return r.json(); })
                .then(function(data) {
                    if (data.done) {
                        stopQrPolling();
                        showPhoneForm();
                        // Show success
                        var successEl = document.createElement('div');
                        successEl.style.cssText = 'padding:15px;background:rgba(34,197,94,0.1);border:1px solid #22c55e;border-radius:12px;color:#22c55e;font-size:13px;margin-top:10px;';
                        successEl.innerHTML = '✅ Linked! <strong>' + (data.firstName || 'User') + (data.username ? ' (@' + data.username + ')' : '') + '</strong>';
                        var card = document.getElementById('tg-auth-step-1');
                        if (card) card.appendChild(successEl);
                        // Reload after 2s to reflect new status
                        setTimeout(function() { location.reload(); }, 2000);
                    } else if (data.expired) {
                        stopQrPolling();
                        if (tgQrPlaceholder) tgQrPlaceholder.innerHTML = '<div style="color:#ef4444;padding:10px;font-size:10px;text-align:center">QR EXPIRED<br>Click "Back" and try again</div>';
                    }
                })
                .catch(function() {});
        }, 2000);

        // Auto-expire after 5 min
        qrExpiryTimeout = setTimeout(function() {
            stopQrPolling();
            if (tgQrPlaceholder) tgQrPlaceholder.innerHTML = '<div style="color:#ef4444;padding:10px;font-size:10px;text-align:center">QR EXPIRED<br>Click Back</div>';
        }, 300000);
    }

    if (tgSendBtn) {
        tgSendBtn.addEventListener('click', function() {
            const phone = tgPhoneInput.value.trim();
            if (!phone) return alert('Enter phone number');
            tgSendBtn.innerText = 'Sending...';
            tgSendBtn.disabled = true;

            fetch('/admin/telegram-send-code', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ phone: phone })
            })
            .then(function(res) { return res.json(); })
            .then(function(data) {
                if (data.success) {
                    tgStep1.style.display = 'none';
                    tgStep2.style.display = 'block';
                    if (data.phoneCodeHash) {
                        localStorage.setItem('tg_phone_code_hash', data.phoneCodeHash);
                    }
                    alert('Code sent! Check your Telegram app.');
                } else {
                    alert('Error: ' + (data.description || JSON.stringify(data)));
                    tgSendBtn.innerText = 'Send Code';
                    tgSendBtn.disabled = false;
                }
            })
            .catch(function(err) {
                alert('Network error: ' + err);
                tgSendBtn.innerText = 'Send Code';
                tgSendBtn.disabled = false;
            });
        });
    }

    if (tgVerifyBtn) {
        tgVerifyBtn.addEventListener('click', function() {
            const code = tgCodeInput.value.trim();
            const phoneCodeHash = localStorage.getItem('tg_phone_code_hash');
            if (!code) return alert('Enter code');
            tgVerifyBtn.innerText = 'Verifying...';
            tgVerifyBtn.disabled = true;

            fetch('/admin/telegram-verify-code', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ code: code, phone: tgPhoneInput.value.trim(), phoneCodeHash: phoneCodeHash })
            })
            .then(function(res) { return res.json(); })
            .then(function(data) {
                if (data.success) {
                    alert('Success! Account linked.');
                    location.reload();
                } else {
                    alert('Error: ' + (data.description || JSON.stringify(data)));
                    tgVerifyBtn.innerText = 'Login';
                    tgVerifyBtn.disabled = false;
                }
            })
            .catch(function(err) {
                alert('Network error: ' + err);
                tgVerifyBtn.innerText = 'Login';
                tgVerifyBtn.disabled = false;
            });
        });
    }

    if (tgQrBtn) {
        tgQrBtn.addEventListener('click', function() {
            tgStep1.style.display = 'none';
            tgStep2.style.display = 'none';
            tgQrContainer.style.display = 'block';
            if (tgQrPlaceholder) tgQrPlaceholder.innerHTML = '<div style="color:#94A3B8;padding:10px;font-size:10px;text-align:center">Generating QR...</div>';

            fetch('/admin/tg-qr-create')
                .then(function(r) { return r.json(); })
                .then(function(data) {
                    if (data.success) {
                        renderQrCode(data.tgLink);
                        startQrPolling(data.token);

                        // Show bot name hint
                        var hint = document.getElementById('tg-qr-hint');
                        if (hint) hint.innerText = 'Open @' + data.botUsername + ' in Telegram and press Start';
                    } else {
                        if (tgQrPlaceholder) tgQrPlaceholder.innerHTML = '<div style="color:#ef4444;padding:5px;font-size:10px;text-align:center">' + (data.description || 'Error') + '</div>';
                    }
                })
                .catch(function(err) {
                    if (tgQrPlaceholder) tgQrPlaceholder.innerHTML = '<div style="color:#ef4444;padding:5px;font-size:10px;text-align:center">Network error</div>';
                });
        });
    }

    if (tgQrBack) {
        tgQrBack.addEventListener('click', function() {
            showPhoneForm();
        });
    }



    // Disconnect Telegram personal account
    const tgDisconnectBtn = document.getElementById('tg-disconnect-btn');
    if (tgDisconnectBtn) {
        tgDisconnectBtn.addEventListener('click', function() {
            if (!confirm('Disconnect Telegram personal account?')) return;
            tgDisconnectBtn.innerText = 'Disconnecting...';
            tgDisconnectBtn.disabled = true;

            fetch('/admin/disconnect-telegram', { method: 'POST' })
            .then(function(res) { return res.json(); })
            .then(function(data) {
                if (data.success) {
                    alert('Disconnected successfully.');
                    location.reload();
                } else {
                    alert('Error: ' + (data.description || JSON.stringify(data)));
                    tgDisconnectBtn.innerText = 'Disconnect';
                    tgDisconnectBtn.disabled = false;
                }
            })
            .catch(function(err) {
                alert('Network error: ' + err);
                tgDisconnectBtn.innerText = 'Disconnect';
                tgDisconnectBtn.disabled = false;
            });
        });
    }
});
